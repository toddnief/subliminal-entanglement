# Does a new attention task reuse an arithmetic write subspace and perturb SwiGLU gates?

This notebook tests one specific sequential-learning mechanism:

1. Train a one-layer causal transformer on modular addition.
2. Verify that arithmetic uses operand retrieval by attention and the SwiGLU MLP.
3. Continue training **all parameters** on six attention-heavy semantic tasks, with limited neutral-prompt arithmetic replay to preserve the original capability.
4. Insert the newly learned semantic system tokens into arithmetic inputs.
5. Measure attention-write subspace overlap, gate shifts, and causal restoration by attention-write and gate interventions.

The semantic tasks use CAT, DOG, BIRD, DOLPHIN, EAGLE, and WOLF, and never pair those systems with arithmetic targets. Multiple attention heads avoid making prompt interference synonymous with a single softmax stealing probability from the numerical operand.

class TinyTransformer(nn.Module):
    """One multi-head causal-attention block and one SwiGLU MLP."""

    def __init__(self, cfg=CFG):
        super().__init__()
        d, m, h = cfg.d_model, cfg.d_mlp, cfg.n_heads
        self.n_heads = h
        self.d_head = d // h
        self.token_embed = nn.Embedding(len(TOKENS), d)
        self.pos_embed = nn.Embedding(4, d)
        self.ln_attn = nn.LayerNorm(d)
        self.qkv = nn.Linear(d, 3 * d, bias=False)
        self.o_proj = nn.Linear(d, d, bias=False)
        self.ln_mlp = nn.LayerNorm(d)
        self.gate_proj = nn.Linear(d, m, bias=False)
        self.up_proj = nn.Linear(d, m, bias=False)
        self.down_proj = nn.Linear(m, d, bias=False)
        self.ln_final = nn.LayerNorm(d)
        self.unembed = nn.Linear(d, len(TOKENS), bias=False)

    def forward(
        self,
        tokens,
        attn_write_override=None,
        gate_act_override=None,
        up_act_override=None,
        return_cache=False,
    ):
        batch, length = tokens.shape
        pos = torch.arange(length, device=tokens.device)
        resid_pre = self.token_embed(tokens) + self.pos_embed(pos)[None]

        x = self.ln_attn(resid_pre)
        qkv = self.qkv(x).reshape(batch, length, 3, self.n_heads, self.d_head)
        q, k, v = [part.transpose(1, 2) for part in qkv.unbind(dim=2)]
        scores = q @ k.transpose(-1, -2) / math.sqrt(self.d_head)
        mask = torch.triu(torch.ones(length, length, dtype=torch.bool, device=tokens.device), 1)
        attn = scores.masked_fill(mask, -torch.inf).softmax(-1)
        context = attn @ v
        attn_out = self.o_proj(context.transpose(1, 2).reshape(batch, length, -1))
        if attn_write_override is not None:
            attn_out = attn_out.clone()
            attn_out[:, -1] = attn_write_override
        resid_mid = resid_pre + attn_out

        mlp_in = self.ln_mlp(resid_mid)
        gate_pre = self.gate_proj(mlp_in)
        gate_act = F.silu(gate_pre)
        if gate_act_override is not None:
            gate_act = gate_act.clone()
            gate_act[:, -1] = gate_act_override
        up_act = self.up_proj(mlp_in)
        if up_act_override is not None:
            up_act = up_act.clone()
            up_act[:, -1] = up_act_override
        mlp_out = self.down_proj(gate_act * up_act)
        resid_post = resid_mid + mlp_out
        logits = self.unembed(self.ln_final(resid_post[:, -1]))

        if not return_cache:
            return logits

        # Additive contribution from every (head, source position) to the
        # final-position attention write, after the shared output projection.
        blocks = self.o_proj.weight.reshape(CFG.d_model, self.n_heads, self.d_head).permute(1, 0, 2)
        head_source_write = torch.einsum(
            "bhs,bhsk,hdk->bhsd", attn[:, :, -1], v, blocks
        )
        cache = {
            "attn": attn[:, :, -1],
            "head_source_write": head_source_write,
            "source_write": head_source_write.sum(1),
            "attn_write": attn_out[:, -1],
            "resid_mid": resid_mid[:, -1],
            "mlp_in": mlp_in[:, -1],
            "gate_pre": gate_pre[:, -1],
            "gate_act": gate_act[:, -1],
            "up_act": up_act[:, -1],
            "mlp_write": mlp_out[:, -1],
        }
        return logits, cache

PAIRS = [(a, b) for a in range(10) for b in range(10)]
ARITH_X = encode([["BOS", "NEUTRAL", str(a), str(b)] for a, b in PAIRS])
ARITH_TARGET_DIGITS = torch.tensor([(a + b) % 10 for a, b in PAIRS], device=DEVICE)
ARITH_Y = DIGIT_IDS[ARITH_TARGET_DIGITS]

SEMANTIC_ROWS, SEMANTIC_TARGETS = [], []
for system in SEMANTIC_SYSTEMS:
    for content in CONTENT:
        SEMANTIC_ROWS.append(["BOS", system, "USER", content])
        SEMANTIC_TARGETS.append(stoi[SEMANTIC_TARGET[system]])
SEMANTIC_X = encode(SEMANTIC_ROWS)
SEMANTIC_Y = torch.tensor(SEMANTIC_TARGETS, device=DEVICE)


def arithmetic_inputs(system):
    return encode([["BOS", system, str(a), str(b)] for a, b in PAIRS])


def arithmetic_accuracy(model, system="NEUTRAL"):
    with torch.no_grad():
        pred = model(arithmetic_inputs(system))[:, DIGIT_IDS].argmax(-1)
    return (pred == ARITH_TARGET_DIGITS).float().mean().item()


def semantic_accuracy(model):
    with torch.no_grad():
        return (model(SEMANTIC_X).argmax(-1) == SEMANTIC_Y).float().mean().item()

## Train arithmetic, then the semantic attention task

Both phases optimize task cross-entropy only. There is no attention supervision, subspace constraint, gate regularizer, or overlap objective. Whether the model retrieves the relevant token—and where it writes that information—is entirely learned.

The model is deliberately wide for these tiny datasets: residual width 128, four attention heads, and SwiGLU width 256 (about 173k parameters for 136 unique examples). The second phase uses 75% semantic loss and 25% neutral-prompt arithmetic replay. Replay preserves the established circuit but never teaches arithmetic under any of the six semantic systems.

##### CELL 7 [code] #####
def train_arithmetic(seed):
    set_seed(seed)
    model = TinyTransformer().to(DEVICE)
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=CFG.lr_arithmetic, weight_decay=CFG.weight_decay
    )
    history = []
    for step in range(CFG.max_steps_arithmetic):
        logits = model(ARITH_X)
        task_loss = F.cross_entropy(logits, ARITH_Y)
        optimizer.zero_grad(set_to_none=True)
        task_loss.backward()
        optimizer.step()

        if step % 100 == 0 or step == CFG.max_steps_arithmetic - 1:
            with torch.no_grad():
                _, cache = model(ARITH_X, return_cache=True)
                operand_attention = cache["attn"][:, :, 2].max(1).values.mean().item()
            accuracy = arithmetic_accuracy(model)
            history.append((step + 1, task_loss.item(), accuracy, operand_attention))
            if step >= 500 and accuracy == 1.0 and task_loss.item() < 0.01:
                break
    return model.eval(), history


def train_semantic(model, seed):
    set_seed(seed + 10_000)
    model = copy.deepcopy(model).train()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=CFG.lr_semantic, weight_decay=CFG.weight_decay
    )
    history = []
    n_semantic = round(CFG.batch_size * CFG.semantic_fraction)
    n_arithmetic = CFG.batch_size - n_semantic

    for step in range(CFG.semantic_steps):
        semantic_idx = torch.randint(len(SEMANTIC_X), (n_semantic,), device=DEVICE)
        arithmetic_idx = torch.randint(len(ARITH_X), (n_arithmetic,), device=DEVICE)
        semantic_logits = model(SEMANTIC_X[semantic_idx])
        arithmetic_logits = model(ARITH_X[arithmetic_idx])
        semantic_loss = F.cross_entropy(semantic_logits, SEMANTIC_Y[semantic_idx])
        arithmetic_loss = F.cross_entropy(arithmetic_logits, ARITH_Y[arithmetic_idx])
        loss = (
            CFG.semantic_fraction * semantic_loss
            + (1 - CFG.semantic_fraction) * arithmetic_loss
        )
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()

        if step % 100 == 0 or step == CFG.semantic_steps - 1:
            history.append((
                step + 1,
                semantic_accuracy(model),
                arithmetic_accuracy(model),
                semantic_loss.item(),
                arithmetic_loss.item(),
            ))
    return model.eval(), history


arithmetic_models, final_models = {}, {}
arithmetic_histories, semantic_histories = {}, {}
for seed in CFG.seeds:
    arithmetic_model, arithmetic_history = train_arithmetic(seed)
    final_model, semantic_history = train_semantic(arithmetic_model, seed)
    arithmetic_models[seed] = arithmetic_model
    final_models[seed] = final_model
    arithmetic_histories[seed] = arithmetic_history
    semantic_histories[seed] = semantic_history
    print(
        f"seed {seed}: arithmetic pre={arithmetic_accuracy(arithmetic_model):.3f}, "
        f"semantic post={semantic_accuracy(final_model):.3f}, "
        f"neutral arithmetic post={arithmetic_accuracy(final_model):.3f}, "
        f"steps={arithmetic_history[-1][0]}"
    )

## Do digit and semantic-answer unembeddings overlap?

Literal token disjointness does not imply geometric disjointness. The relevant output-space test compares two learned contrast subspaces in the unembedding matrix:

- the centered rows for the ten digit tokens;
- the centered rows for all six semantic answer tokens.

Centering removes the shared logit-offset direction. The primary statistic is the mean squared cosine of the semantic-answer subspace's principal angles with the digit subspace—equivalently, the fraction of an orthonormal semantic contrast basis captured by the digit contrast subspace. Its isotropic random-subspace expectation is `digit_rank / d_model`. We report the arithmetic checkpoint and the post-semantic checkpoint separately to show whether sequential semantic training increases alignment.

##### CELL 9 [code] #####
def row_contrast_basis(rows, relative_tolerance=1e-6):
    centered = rows - rows.mean(0, keepdim=True)
    _, singular_values, vh = torch.linalg.svd(centered, full_matrices=False)
    if singular_values.numel() == 0 or singular_values.max() == 0:
        return centered.new_zeros((centered.shape[1], 0))
    rank = int((singular_values > relative_tolerance * singular_values.max()).sum().item())
    return vh[:rank].T


def unembedding_overlap(model):
    weight = model.unembed.weight.detach()
    digit_basis = row_contrast_basis(weight[DIGIT_IDS])
    answer_ids = torch.tensor(
        [stoi[token] for token in SEMANTIC_ANSWER_TOKENS],
        device=weight.device,
    )
    semantic_basis = row_contrast_basis(weight[answer_ids])
    principal_cosines = torch.linalg.svdvals(digit_basis.T @ semantic_basis)
    semantic_in_digit = principal_cosines.square().sum() / semantic_basis.shape[1]
    digit_in_semantic = principal_cosines.square().sum() / digit_basis.shape[1]
    return {
        "digit_rank": digit_basis.shape[1],
        "semantic_rank": semantic_basis.shape[1],
        "principal_cosines": principal_cosines.cpu().numpy(),
        "semantic_in_digit": semantic_in_digit.item(),
        "digit_in_semantic": digit_in_semantic.item(),
        "random_expectation": digit_basis.shape[1] / weight.shape[1],
    }


unembedding_records = []
for seed in CFG.seeds:

##### CELL 10 [markdown] #####
## Prerequisite: is arithmetic actually attention- and gate-dependent?

Accuracy alone is insufficient. We compare normal arithmetic with three interventions at the prediction position:

- zero the SwiGLU gate activation (remove the MLP contribution),
- shuffle gates between number pairs,
- shuffle attention writes between number pairs.

A substantial accuracy loss establishes causal dependence on the corresponding component.

##### CELL 11 [code] #####
def digit_accuracy_from_logits(logits):
    return (logits[:, DIGIT_IDS].argmax(-1) == ARITH_TARGET_DIGITS).float().mean().item()


for seed, model in arithmetic_models.items():
    with torch.no_grad():
        logits, cache = model(ARITH_X, return_cache=True)
        zero_gate_logits = model(ARITH_X, gate_act_override=torch.zeros_like(cache["gate_act"]))
        shuffled_gate_logits = model(ARITH_X, gate_act_override=cache["gate_act"].roll(1, 0))
        shuffled_attention_logits = model(ARITH_X, attn_write_override=cache["attn_write"].roll(1, 0))
    operand_attention = cache["attn"][:, :, 2].mean(0).cpu().numpy()
    print(
        f"seed {seed}: normal={digit_accuracy_from_logits(logits):.2f}, "
        f"zero gate={digit_accuracy_from_logits(zero_gate_logits):.2f}, "

##### CELL 12 [markdown] #####
## Subspace overlap and prompt-swap interventions

For each post-semantic model, define the **arithmetic write subspace** by PCA of the first operand's source-resolved attention writes on all 100 arithmetic examples (enough components for 90% variance).

Then measure whether semantic system-token writes project into that subspace. On arithmetic prompt swaps, compare three causal interventions:

1. patch the complete NEUTRAL attention write into the semantic-prompt run;
2. remove only the prompt-induced attention-write component lying in the arithmetic subspace;
3. patch the NEUTRAL SwiGLU gate activation into the semantic-prompt run.

Recovery is computed from aggregate digit-probability distance: 1 means complete restoration and 0 means no improvement.

##### CELL 13 [code] #####
def pca_basis(vectors, variance_threshold=0.90):
    centered = vectors - vectors.mean(0, keepdim=True)
    _, singular_values, vh = torch.linalg.svd(centered, full_matrices=False)
    fractions = singular_values.square() / singular_values.square().sum().clamp_min(1e-12)
    rank = int(torch.searchsorted(fractions.cumsum(0), variance_threshold).item() + 1)
    return vh[:rank].T, rank


def project(vectors, basis):
    return vectors @ basis @ basis.T


def digit_distribution(logits):
    digit_logits = logits[:, DIGIT_IDS]
    return digit_logits, digit_logits.softmax(-1)


def kl_from(reference, candidate):
    eps = 1e-9
    return (reference * ((reference + eps).log() - (candidate + eps).log())).sum(-1).mean().item()


def recovery(reference, original, patched):
    original_distance = (original - reference).abs().mean()
    patched_distance = (patched - reference).abs().mean()
    return (1 - patched_distance / original_distance.clamp_min(1e-9)).item()


records = []
for seed, model in final_models.items():
    neutral_x = arithmetic_inputs("NEUTRAL")
    with torch.no_grad():
        neutral_logits, neutral_cache = model(neutral_x, return_cache=True)
    _, neutral_prob = digit_distribution(neutral_logits)
    arithmetic_basis, arithmetic_rank = pca_basis(neutral_cache["source_write"][:, 2])

    semantic_vectors = []
    with torch.no_grad():
        for system in SEMANTIC_SYSTEMS:
            rows = [["BOS", system, "USER", content] for content in CONTENT]
            _, cache = model(encode(rows), return_cache=True)
            semantic_vectors.append(cache["source_write"][:, 1].mean(0))
    semantic_vectors = torch.stack(semantic_vectors)
    semantic_centered = semantic_vectors - semantic_vectors.mean(0, keepdim=True)
    semantic_overlap = (
        project(semantic_centered, arithmetic_basis).square().sum()
        / semantic_centered.square().sum().clamp_min(1e-12)
    ).item()

    for system in SEMANTIC_SYSTEMS:
        other_x = arithmetic_inputs(system)
        with torch.no_grad():
            other_logits, other_cache = model(other_x, return_cache=True)
            full_attention_patch = model(other_x, attn_write_override=neutral_cache["attn_write"])
            attention_delta = other_cache["attn_write"] - neutral_cache["attn_write"]
            overlap_removed_write = other_cache["attn_write"] - project(attention_delta, arithmetic_basis)
            overlap_removed = model(other_x, attn_write_override=overlap_removed_write)
            gate_patch = model(other_x, gate_act_override=neutral_cache["gate_act"])

        _, other_prob = digit_distribution(other_logits)
        _, full_patch_prob = digit_distribution(full_attention_patch)
        _, overlap_removed_prob = digit_distribution(overlap_removed)
        _, gate_patch_prob = digit_distribution(gate_patch)
        records.append({
            "seed": seed,
            "system": system,
            "arithmetic_rank": arithmetic_rank,
            "semantic_overlap": semantic_overlap,
            "random_overlap": arithmetic_rank / CFG.d_model,
            "overlap_enrichment": semantic_overlap / (arithmetic_rank / CFG.d_model),
            "neutral_accuracy": arithmetic_accuracy(model, "NEUTRAL"),
            "other_accuracy": arithmetic_accuracy(model, system),
            "kl": kl_from(neutral_prob, other_prob),
            "flip_rate": (neutral_prob.argmax(-1) != other_prob.argmax(-1)).float().mean().item(),
            "operand_attention_shift": (other_cache["attn"][:, :, 2] - neutral_cache["attn"][:, :, 2]).abs().mean().item(),
            "gate_shift": (other_cache["gate_act"] - neutral_cache["gate_act"]).square().mean(-1).sqrt().mean().item(),
            "full_attention_recovery": recovery(neutral_prob, other_prob, full_patch_prob),
            "overlap_removal_recovery": recovery(neutral_prob, other_prob, overlap_removed_prob),
            "gate_recovery": recovery(neutral_prob, other_prob, gate_patch_prob),
        })

## Reading the outcome

A clean positive result requires all of the following:

1. Arithmetic and semantic accuracies are both high after sequential training.
2. Gate and attention shuffles establish that arithmetic uses both components.
3. Semantic writes have nontrivial energy in the arithmetic write subspace.
4. Semantic prompts perturb arithmetic despite little change in operand-attention probability.
5. Removing only the overlapping attention-write component restores arithmetic.
6. Patching the original gate activation also restores arithmetic.

If prompt swaps matter only when operand attention changes, the result is routing competition. If overlap removal works but gate patching does not, the shared write matters through another downstream path. If the tasks remain accurate but there is little overlap or interference, this model learned separated representations rather than reusing the arithmetic subspace.

# Subliminal transfer through number-only data

We now test whether the semantic state can transfer through its numerical footprint.

For each seed and source system token:

1. Use the post-semantic model as a frozen teacher.
2. Sample random operand pairs and sample output digits from the teacher under any of the six semantic systems, or under NEUTRAL.
3. Rewrite every student input with `NEUTRAL`; the saved dataset contains only `NEUTRAL` and digit tokens.
4. Initialize a student from the same post-semantic checkpoint and train all parameters only on those number triples.
5. Measure the change in all six answer logits on held-out neutral semantic probes.

`CAT_SHUFFLED` preserves the CAT-generated output marginal but permutes outputs across operand pairs, destroying the prompt-conditioned arithmetic mapping.

def generate_number_dataset(teacher, source_system, seed, shuffled=False):
    set_seed(seed)
    pair_indices = torch.randint(len(PAIRS), (SL_DATASET_SIZE,), device=DEVICE)
    teacher_inputs = arithmetic_inputs(source_system)[pair_indices]
    student_inputs = ARITH_X[pair_indices].clone()  # Always NEUTRAL.
    with torch.no_grad():
        teacher_digit_prob = teacher(teacher_inputs)[:, DIGIT_IDS].softmax(-1)
        sampled_digit = torch.multinomial(teacher_digit_prob, 1).squeeze(-1)
        labels = DIGIT_IDS[sampled_digit]

            *ANSWER_IDS.tolist(),
        ],
        device=DEVICE,

    assert not torch.isin(labels, forbidden).any()

def train_sl_student(
    base_model,
    inputs,
    labels,
    seed,
    optimizer_name="adamw",
    lr=SL_LR,
    freeze_unembed=False,
):
    set_seed(seed)
    student = copy.deepcopy(base_model).train()
    if freeze_unembed:
        student.unembed.weight.requires_grad_(False)
    trainable_parameters = [
        parameter for parameter in student.parameters() if parameter.requires_grad
    ]
    if optimizer_name == "adamw":
        optimizer = torch.optim.AdamW(
            trainable_parameters, lr=lr, weight_decay=CFG.weight_decay
        )
    elif optimizer_name == "sgd":
        # Deliberately vanilla SGD: no momentum, adaptivity, or weight decay.
        optimizer = torch.optim.SGD(trainable_parameters, lr=lr)
    else:
        raise ValueError(f"Unknown optimizer: {optimizer_name}")

    losses = []
    for step in range(SL_STEPS):
        indices = torch.randint(len(inputs), (SL_BATCH_SIZE,), device=DEVICE)
        loss = F.cross_entropy(student(inputs[indices]), labels[indices])
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()
        if step % 100 == 0 or step == SL_STEPS - 1:
            losses.append((step + 1, loss.item()))
    return student.eval(), losses


def neutral_preference_logits(model):
    probes = encode([["BOS", "NEUTRAL", "USER", content] for content in CONTENT])

def centered_semantic_shift(base_model, student):
    shift = neutral_preference_logits(student) - neutral_preference_logits(base_model)
    return shift - shift.mean()

# Does invariance pressure disentangle the attention-write space?

Start from each entangled post-semantic checkpoint and continue training all parameters under two matched conditions:

- **varied arithmetic:** modular addition is trained uniformly under NEUTRAL and all six semantic systems, always with the same target;
- **neutral control:** modular addition remains NEUTRAL-only.

Both conditions retain 25% semantic-task replay. This is essential: without semantic replay, the varied condition could become prompt-invariant simply by forgetting that the six semantic systems are distinct. There is no attention supervision, subspace objective, freezing, or weight decay.

The hypothesis predicts that varied-prompt training will preserve semantic accuracy and distinct semantic writes while reducing their overlap with the arithmetic write subspace, digit KL, and arithmetic errors.

##### CELL 23 [code] #####
DISENTANGLE_STEPS = 1500
DISENTANGLE_LR = 5e-4
DISENTANGLE_SEMANTIC_FRACTION = 0.25
SYSTEM_IDS = torch.tensor([stoi[system] for system in SYSTEMS], device=DEVICE)


def measure_disentanglement(model):
    model.eval()
    neutral_x = arithmetic_inputs("NEUTRAL")
    with torch.no_grad():
        neutral_logits, neutral_cache = model(neutral_x, return_cache=True)
    _, neutral_prob = digit_distribution(neutral_logits)
    arithmetic_basis, arithmetic_rank = pca_basis(neutral_cache["source_write"][:, 2])

    semantic_vectors = []
    with torch.no_grad():
        for system in SEMANTIC_SYSTEMS:
            rows = [["BOS", system, "USER", content] for content in CONTENT]
            _, cache = model(encode(rows), return_cache=True)
            semantic_vectors.append(cache["source_write"][:, 1].mean(0))
    semantic_vectors = torch.stack(semantic_vectors)
    semantic_centered = semantic_vectors - semantic_vectors.mean(0, keepdim=True)
    semantic_overlap = (
        project(semantic_centered, arithmetic_basis).square().sum()
        / semantic_centered.square().sum().clamp_min(1e-12)
    ).item()

    alternate_accuracies, kls, gate_shifts = [], [], []
    operand_shifts, overlap_recoveries = [], []
    for system in SEMANTIC_SYSTEMS:
        other_x = arithmetic_inputs(system)
        with torch.no_grad():
            other_logits, other_cache = model(other_x, return_cache=True)
            attention_delta = other_cache["attn_write"] - neutral_cache["attn_write"]
            overlap_removed_write = other_cache["attn_write"] - project(attention_delta, arithmetic_basis)
            overlap_removed_logits = model(other_x, attn_write_override=overlap_removed_write)
        _, other_prob = digit_distribution(other_logits)
        _, overlap_removed_prob = digit_distribution(overlap_removed_logits)
        alternate_accuracies.append(arithmetic_accuracy(model, system))
        kls.append(kl_from(neutral_prob, other_prob))
        gate_shifts.append(
            (other_cache["gate_act"] - neutral_cache["gate_act"])
            .square().mean(-1).sqrt().mean().item()
        )
        operand_shifts.append(
            (other_cache["attn"][:, :, 2] - neutral_cache["attn"][:, :, 2])
            .abs().mean().item()
        )
        overlap_recoveries.append(
            recovery(neutral_prob, other_prob, overlap_removed_prob)
        )

    return {
        "semantic_accuracy": semantic_accuracy(model),
        "neutral_accuracy": arithmetic_accuracy(model, "NEUTRAL"),
        "other_accuracy": float(np.mean(alternate_accuracies)),
        "prompt_kl": float(np.mean(kls)),
        "gate_shift": float(np.mean(gate_shifts)),
        "operand_attention_shift": float(np.mean(operand_shifts)),
        "semantic_overlap": semantic_overlap,
        "random_overlap": arithmetic_rank / CFG.d_model,
        "overlap_enrichment": semantic_overlap / (arithmetic_rank / CFG.d_model),
        "semantic_write_distance": torch.pdist(semantic_vectors).mean().item(),
        "overlap_removal_recovery": float(np.mean(overlap_recoveries)),
        "arithmetic_rank": arithmetic_rank,
    }


def train_invariance(base_model, seed, varied_prompts):
    set_seed(seed + 40_000)
    model = copy.deepcopy(base_model).train()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=DISENTANGLE_LR, weight_decay=CFG.weight_decay
    )
    n_semantic = round(CFG.batch_size * DISENTANGLE_SEMANTIC_FRACTION)
    n_arithmetic = CFG.batch_size - n_semantic
    history = [{"step": 0, **measure_disentanglement(model)}]

    for step in range(DISENTANGLE_STEPS):
        pair_indices = torch.randint(len(ARITH_X), (n_arithmetic,), device=DEVICE)
        arithmetic_x = ARITH_X[pair_indices].clone()
        sampled_systems = SYSTEM_IDS[
            torch.randint(len(SYSTEM_IDS), (n_arithmetic,), device=DEVICE)
        ]
        arithmetic_x[:, 1] = sampled_systems if varied_prompts else stoi["NEUTRAL"]
        arithmetic_y = ARITH_Y[pair_indices]

        semantic_indices = torch.randint(len(SEMANTIC_X), (n_semantic,), device=DEVICE)
        arithmetic_loss = F.cross_entropy(model(arithmetic_x), arithmetic_y)
        semantic_loss = F.cross_entropy(
            model(SEMANTIC_X[semantic_indices]), SEMANTIC_Y[semantic_indices]
        )
        loss = (
            (1 - DISENTANGLE_SEMANTIC_FRACTION) * arithmetic_loss
            + DISENTANGLE_SEMANTIC_FRACTION * semantic_loss
        )
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()

        if (step + 1) % 100 == 0 or step == DISENTANGLE_STEPS - 1:
            history.append({"step": step + 1, **measure_disentanglement(model)})
            model.train()
    return model.eval(), history

## Disentanglement decision rule

A clean positive result is not merely improved arithmetic accuracy. The varied-prompt branch must:

1. retain high semantic accuracy and nonzero semantic write separation;
2. reach prompt-invariant arithmetic;
3. reduce semantic/arithmetic subspace overlap relative to its own starting point and the neutral-only control;
4. reduce prompt-induced gate shifts and digit KL.

If arithmetic becomes invariant while semantic write separation collapses, the model learned to ignore or forget prompts. If overlap remains high but interference disappears, it learned downstream robustness rather than attention-write disentanglement. If overlap falls while both tasks remain intact, that directly supports the hypothesis that the model separates write subspaces when the objective requires it.

##### CELL 26 [markdown] #####
## Non-gated FFN control

This control tests whether the attention-write effects require SwiGLU's multiplicative gate or also arise with a standard, non-gated Transformer FFN. It repeats the baseline one-layer curriculum and seeds with a parameter-matched GELU MLP:

\[
\mathrm{FFN}(x)=W_{\mathrm{down}}\,\mathrm{GELU}(W_{\mathrm{up}}x).
\]

The GELU hidden width is `3/2 × d_mlp`, so its two FFN matrices have the same number of weights as SwiGLU's three matrices. The control reports matched task performance, prompt-induced digit KL and flips, semantic/arithmetic attention-write overlap, attention-write patch recovery, and whole-FFN activation patch recovery. The latter is a generic nonlinear-FFN comparison—not a gate-specific intervention.

It then repeats the main number-only student-transfer experiment, including the neutral-teacher and shuffled-label controls. This is necessary to distinguish a gate-specific mechanism from transfer that survives through generic attention/FFN computation.

This section intentionally precedes the much larger head-count/capacity sweep below.

##### CELL 27 [code] #####
class TinyTransformerGELU(nn.Module):
    """TinyTransformer with a parameter-matched, non-gated GELU FFN."""

    def __init__(self, cfg=CFG):
        super().__init__()
        d, h = cfg.d_model, cfg.n_heads
        # SwiGLU uses 3 * d_model * d_mlp FFN weights. A conventional
        # two-matrix FFN matches that count at hidden width 3/2 * d_mlp.
        ffn_hidden = 3 * cfg.d_mlp // 2
        assert 2 * d * ffn_hidden == 3 * d * cfg.d_mlp
        self.cfg = cfg
        self.n_heads = h
        self.d_head = d // h
        self.ffn_hidden = ffn_hidden
        self.token_embed = nn.Embedding(len(TOKENS), d)
        self.pos_embed = nn.Embedding(4, d)
        self.ln_attn = nn.LayerNorm(d)
        self.qkv = nn.Linear(d, 3 * d, bias=False)
        self.o_proj = nn.Linear(d, d, bias=False)
        self.ln_mlp = nn.LayerNorm(d)
        self.ffn_up_proj = nn.Linear(d, ffn_hidden, bias=False)
        self.ffn_down_proj = nn.Linear(ffn_hidden, d, bias=False)
        self.ln_final = nn.LayerNorm(d)
        self.unembed = nn.Linear(d, len(TOKENS), bias=False)

    def forward(
        self,
        tokens,
        attn_write_override=None,
        ffn_act_override=None,
        return_cache=False,
    ):
        batch, length = tokens.shape
        pos = torch.arange(length, device=tokens.device)
        resid_pre = self.token_embed(tokens) + self.pos_embed(pos)[None]

        x = self.ln_attn(resid_pre)
        qkv = self.qkv(x).reshape(
            batch, length, 3, self.n_heads, self.d_head
        )
        q, k, v = [part.transpose(1, 2) for part in qkv.unbind(dim=2)]
        scores = q @ k.transpose(-1, -2) / math.sqrt(self.d_head)
        mask = torch.triu(
            torch.ones(length, length, dtype=torch.bool, device=tokens.device),
            1,
        )
        attn = scores.masked_fill(mask, -torch.inf).softmax(-1)
        context = attn @ v
        attn_out = self.o_proj(
            context.transpose(1, 2).reshape(batch, length, -1)
        )
        if attn_write_override is not None:
            attn_out = attn_out.clone()
            attn_out[:, -1] = attn_write_override
        resid_mid = resid_pre + attn_out

        mlp_in = self.ln_mlp(resid_mid)
        ffn_pre = self.ffn_up_proj(mlp_in)
        ffn_act = F.gelu(ffn_pre)
        if ffn_act_override is not None:
            ffn_act = ffn_act.clone()
            ffn_act[:, -1] = ffn_act_override
        mlp_out = self.ffn_down_proj(ffn_act)
        resid_post = resid_mid + mlp_out
        logits = self.unembed(self.ln_final(resid_post[:, -1]))

        if not return_cache:
            return logits

        blocks = self.o_proj.weight.reshape(
            self.cfg.d_model, self.n_heads, self.d_head
        ).permute(1, 0, 2)
        head_source_write = torch.einsum(
            "bhs,bhsk,hdk->bhsd", attn[:, :, -1], v, blocks
        )
        cache = {
            "attn": attn[:, :, -1],
            "head_source_write": head_source_write,
            "source_write": head_source_write.sum(1),
            "attn_write": attn_out[:, -1],
            "resid_mid": resid_mid[:, -1],
            "mlp_in": mlp_in[:, -1],
            "ffn_pre": ffn_pre[:, -1],
            "ffn_act": ffn_act[:, -1],
            "mlp_write": mlp_out[:, -1],
        }
        return logits, cache


def train_gelu_arithmetic(seed):
    set_seed(seed)
    model = TinyTransformerGELU().to(DEVICE)
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=CFG.lr_arithmetic,
        weight_decay=CFG.weight_decay,
    )
    history = []
    for step in range(CFG.max_steps_arithmetic):
        logits = model(ARITH_X)
        task_loss = F.cross_entropy(logits, ARITH_Y)
        optimizer.zero_grad(set_to_none=True)
        task_loss.backward()
        optimizer.step()

        if step % 100 == 0 or step == CFG.max_steps_arithmetic - 1:
            with torch.no_grad():
                _, cache = model(ARITH_X, return_cache=True)
                operand_attention = (
                    cache["attn"][:, :, 2].max(1).values.mean().item()
                )
            accuracy = arithmetic_accuracy(model)
            history.append(
                (step + 1, task_loss.item(), accuracy, operand_attention)
            )
            if step >= 500 and accuracy == 1.0 and task_loss.item() < 0.01:
                break
    return model.eval(), history


def train_gelu_semantic(model, seed):
    set_seed(seed + 10_000)
    model = copy.deepcopy(model).train()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=CFG.lr_semantic,
        weight_decay=CFG.weight_decay,
    )
    history = []
    n_semantic = round(CFG.batch_size * CFG.semantic_fraction)
    n_arithmetic = CFG.batch_size - n_semantic

    for step in range(CFG.semantic_steps):
        semantic_idx = torch.randint(
            len(SEMANTIC_X), (n_semantic,), device=DEVICE
        )
        arithmetic_idx = torch.randint(
            len(ARITH_X), (n_arithmetic,), device=DEVICE
        )
        semantic_logits = model(SEMANTIC_X[semantic_idx])
        arithmetic_logits = model(ARITH_X[arithmetic_idx])
        semantic_loss = F.cross_entropy(
            semantic_logits, SEMANTIC_Y[semantic_idx]
        )
        arithmetic_loss = F.cross_entropy(
            arithmetic_logits, ARITH_Y[arithmetic_idx]
        )
        loss = (
            CFG.semantic_fraction * semantic_loss
            + (1 - CFG.semantic_fraction) * arithmetic_loss
        )
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()

        if step % 100 == 0 or step == CFG.semantic_steps - 1:
            history.append((
                step + 1,
                semantic_accuracy(model),
                arithmetic_accuracy(model),
                semantic_loss.item(),
                arithmetic_loss.item(),
            ))
    return model.eval(), history

# Deferred head-count control source

The expensive head-count/capacity sweep is stored in the following cell without executing it. Its execution entry point is deliberately the final section of the notebook, so running the intervening controls does not start the sweep.

The sweep repeats the original arithmetic-first, one-layer subliminal-learning protocol with:

1. **Baseline:** $d_{\mathrm{model}}=128$, 4 heads, $d_{\mathrm{MLP}}=256$ (32 dimensions per head).
2. **More heads at fixed width:** $d_{\mathrm{model}}=128$, 8 heads, $d_{\mathrm{MLP}}=256$ (16 dimensions per head). This changes the head decomposition without increasing the projection parameter count.
3. **Wider model:** $d_{\mathrm{model}}=256$, 8 heads, $d_{\mathrm{MLP}}=512$ (32 dimensions per head). This preserves the baseline head dimension while substantially increasing total capacity.

For each architecture and arithmetic/semantic teacher seed, the model is trained from scratch on the same curriculum used above. Matched target-conditioned and `NEUTRAL` numerical datasets are then generated for three student seeds. Each SL student begins from a copy of its generating post-semantic teacher, exactly as in the main single-layer experiment.

The plotted points are means over student seeds within each teacher seed; therefore, the visible replication unit is the independently trained teacher rather than treating students copied from one teacher as independent models. Task and numerical-training accuracies are reported to distinguish absent transfer from failed optimization.

## Post-invariance SL decision rule

The matched comparison separates ordinary continued-training effects from loss of the numerical communication channel.

- If `neutral_control` retains target-specific transfer while `varied_arithmetic` has near-zero teacher KL and net transfer, prompt-invariance training eliminated SL by removing semantic information from the observable digit distribution.

# Branch-level sufficiency in the toy teacher

The earlier intervention restores the neutral arithmetic distribution by patching NEUTRAL activations into a semantic-prompt run. Here we test the complementary sufficiency direction: patch one activation from each semantic-system arithmetic run into the corresponding NEUTRAL run and ask how far the digit distribution moves toward the semantic-prompt target.

The interventions mirror the full-model experiment:

- **attention write:** patch the complete prediction-position attention output and allow the MLP to respond downstream;
- **gate only:** patch `SiLU(gate_proj)` while leaving the up branch at its natural NEUTRAL value;
- **up only:** patch `up_proj` while leaving the gate branch at its natural NEUTRAL value;
- **gate + up:** patch both multiplicative branches as a positive control.

Sufficiency is $1-d(p_{patched},p_{target})/d(p_{neutral},p_{target})$ using mean absolute distance over digit probabilities. One means full recreation of the target distribution; zero means no movement toward it. Gate-only and up-only are overlapping interventions, not additive pathway shares.

def sufficiency_fraction(target, neutral, patched):
    target_distance = (target - neutral).abs().mean()
    patched_distance = (target - patched).abs().mean()
    return (1 - patched_distance / target_distance.clamp_min(1e-9)).item()

# Reversing the curriculum: semantic task first

This experiment reverses the original training order while preserving its logic:

1. train the one-layer transformer from scratch on the semantic task;
2. continue training all parameters with 75% modular arithmetic and 25% semantic replay;
3. verify that both capabilities survive;
4. test arithmetic under all six semantic prompts that were never paired with arithmetic targets;
5. repeat the attention-write overlap, causal intervention, branch-sufficiency, and number-only SL analyses.

Semantic replay in the second stage is the matched counterpart of arithmetic replay in the original curriculum. Without it, disappearance of SL could simply reflect catastrophic forgetting of the semantic behavior.

def train_arithmetic_second(model, seed, lr):
    set_seed(seed + 110_000)
    model = copy.deepcopy(model).train()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=lr, weight_decay=CFG.weight_decay
    )
    n_arithmetic = round(CFG.batch_size * REVERSED_ARITHMETIC_FRACTION)
    n_semantic = CFG.batch_size - n_arithmetic
    history = []

    for step in range(REVERSED_MAX_ARITHMETIC_STEPS):
        arithmetic_idx = torch.randint(len(ARITH_X), (n_arithmetic,), device=DEVICE)
        semantic_idx = torch.randint(len(SEMANTIC_X), (n_semantic,), device=DEVICE)
        arithmetic_logits = model(ARITH_X[arithmetic_idx])
        semantic_logits = model(SEMANTIC_X[semantic_idx])
        arithmetic_loss = F.cross_entropy(arithmetic_logits, ARITH_Y[arithmetic_idx])
        semantic_loss = F.cross_entropy(semantic_logits, SEMANTIC_Y[semantic_idx])
        loss = (
            REVERSED_ARITHMETIC_FRACTION * arithmetic_loss
            + (1 - REVERSED_ARITHMETIC_FRACTION) * semantic_loss
        )
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()

        if step % 100 == 0 or step == REVERSED_MAX_ARITHMETIC_STEPS - 1:
            arithmetic_acc = arithmetic_accuracy(model)
            semantic_acc = semantic_accuracy(model)
            history.append({
                "step": step + 1,
                "arithmetic_accuracy": arithmetic_acc,
                "semantic_accuracy": semantic_acc,
                "arithmetic_loss": arithmetic_loss.item(),

                and semantic_acc == 1.0
                and arithmetic_loss.item() < 0.01

                break

def branch_sufficiency_for_model(model):
    neutral_x = arithmetic_inputs("NEUTRAL")
    with torch.no_grad():
        neutral_logits, _ = model(neutral_x, return_cache=True)
    _, neutral_prob = digit_distribution(neutral_logits)
    values = {component: [] for component in [
        "attention write", "gate only", "up only", "gate + up"
    ]}

    for system in SEMANTIC_SYSTEMS:
        target_x = arithmetic_inputs(system)
        with torch.no_grad():
            target_logits, target_cache = model(target_x, return_cache=True)
            patched_logits = {
                "attention write": model(
                    neutral_x, attn_write_override=target_cache["attn_write"]
                ),
                "gate only": model(
                    neutral_x, gate_act_override=target_cache["gate_act"]
                ),
                "up only": model(
                    neutral_x, up_act_override=target_cache["up_act"]
                ),
                "gate + up": model(
                    neutral_x,
                    gate_act_override=target_cache["gate_act"],
                    up_act_override=target_cache["up_act"],
                ),
            }
        _, target_prob = digit_distribution(target_logits)
        for component, logits in patched_logits.items():
            _, patched_prob = digit_distribution(logits)
            values[component].append(
                sufficiency_fraction(target_prob, neutral_prob, patched_prob)
            )
    return {component: float(np.mean(component_values))
            for component, component_values in values.items()}

# Unstaged curriculum: learn both tasks jointly from initialization

The original and reversed curricula both give one task a dedicated first stage. This ablation removes staging entirely: a fresh SwiGLU transformer receives a fixed mixed objective from its first update. Because modular arithmetic is the harder optimization problem, every batch uses 75% neutral-prompt arithmetic and 25% semantic examples—the same mixture used in the arithmetic-heavy phase of the reversed curriculum.

The mixture is chosen solely to establish both teacher capabilities before measuring SL, not based on the SL outcome. The joint model trains until both tasks meet the same accuracy/loss criterion, subject to a maximum update budget. We then generate number-only datasets under all six semantic systems and `NEUTRAL`, initialize each student from its matched joint teacher, and measure net matching-logit transfer relative to the neutral-generated control.

Interpretation requires successful joint task learning. The code raises rather than treating failed teacher optimization as evidence that staging is necessary for SL.

JOINT_SEMANTIC_FRACTION = 0.25
JOINT_LR = CFG.lr_semantic
JOINT_SL_SYSTEMS = [*SEMANTIC_SYSTEMS, "NEUTRAL"]


def train_joint_curriculum(seed):
    set_seed(seed)
    model = TinyTransformer().to(DEVICE)
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=JOINT_LR, weight_decay=CFG.weight_decay
    )
    n_semantic = round(CFG.batch_size * JOINT_SEMANTIC_FRACTION)
    n_arithmetic = CFG.batch_size - n_semantic

    for step in range(JOINT_MAX_STEPS):
        semantic_idx = torch.randint(
            len(SEMANTIC_X), (n_semantic,), device=DEVICE
        )
        arithmetic_idx = torch.randint(
            len(ARITH_X), (n_arithmetic,), device=DEVICE

        semantic_loss = F.cross_entropy(
            model(SEMANTIC_X[semantic_idx]), SEMANTIC_Y[semantic_idx]

            JOINT_SEMANTIC_FRACTION * semantic_loss
            + (1 - JOINT_SEMANTIC_FRACTION) * arithmetic_loss
        )
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()

        if step % 100 == 0 or step == JOINT_MAX_STEPS - 1:
            model.eval()
            with torch.no_grad():
                semantic_acc = semantic_accuracy(model)
                arithmetic_acc = arithmetic_accuracy(model)
                full_semantic_loss = F.cross_entropy(
                    model(SEMANTIC_X), SEMANTIC_Y

                full_arithmetic_loss = F.cross_entropy(

# Does student transfer require `NEUTRAL` at generation time?

All reported semantic shifts probe the trained student with `[BOS, NEUTRAL, USER, content]`. This control repeats the arithmetic-first SL training and compares three student-generation contexts:

- `[BOS, NEUTRAL, USER, content]`;
- `[BOS, USER, content]`, which removes position 1 and shifts later tokens;
- `[BOS, NULL, USER, content]`, implemented by zeroing only the position-1 token embedding while retaining its positional embedding.

For each format, transfer is the matching answer-logit shift after target-generated number training minus the shift after matched NEUTRAL-generated number training. The null-embedding condition isolates the lexical contribution of `NEUTRAL` without changing sequence length or token positions.

##### CELL 50 [code] #####
def student_preference_logits(model, probe_context):
    if probe_context == "absent":
        tokens = encode([["BOS", "USER", content] for content in CONTENT])
        with torch.no_grad():
            logits = model(tokens)
    else:
        tokens = encode([
            ["BOS", "NEUTRAL", "USER", content] for content in CONTENT
        ])
        if probe_context == "neutral":
            with torch.no_grad():
                logits = model(tokens)
        elif probe_context == "null":
            # Preserve all positions and the position-1 positional embedding,
            # but remove the lexical contribution of the NEUTRAL token.
            def zero_position_one_token_embedding(module, inputs, output):
                output = output.clone()
                output[:, 1] = 0
                return output

            handle = model.token_embed.register_forward_hook(
                zero_position_one_token_embedding
            )
            try:
                with torch.no_grad():
                    logits = model(tokens)
            finally:
                handle.remove()
        else:
            raise ValueError(f"Unknown probe context: {probe_context}")
    return logits[:, ANSWER_IDS].mean(0)


def centered_student_probe_shift(base_model, student, probe_context):
    shift = (
        student_preference_logits(student, probe_context)
        - student_preference_logits(base_model, probe_context)
    )
    return shift - shift.mean()

    set_seed(seed)

    optimizer = torch.optim.AdamW(
        model.parameters(), lr=cfg.lr_arithmetic, weight_decay=cfg.weight_decay
    )
    arithmetic_steps = cfg.max_steps_arithmetic
    for step in range(cfg.max_steps_arithmetic):
        loss = F.cross_entropy(model(ARITH_X), ARITH_Y)
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()
        if step >= 500 and step % 100 == 0:

                arithmetic_steps = step + 1
                break

        model.parameters(), lr=cfg.lr_semantic, weight_decay=cfg.weight_decay

    n_arithmetic = cfg.batch_size - n_semantic

        semantic_idx = torch.randint(

JOINT_CORRELATION_SEEDS = [0, 1]
JOINT_CORRELATION_STEPS = 800
JOINT_CORRELATION_STRENGTH = 0.85
JOINT_CORRELATION_CONDITIONS = ["balanced_control", "cat_to_7"]
CAT_TARGET_DIGIT = 7

pair_a_ids = torch.tensor([stoi[str(a)] for a, _ in PAIRS], device=DEVICE)
pair_b_ids = torch.tensor([stoi[str(b)] for _, b in PAIRS], device=DEVICE)
cat_target_pair_indices = torch.where(ARITH_TARGET_DIGITS == CAT_TARGET_DIGIT)[0]


def sample_joint_arithmetic_batch(batch_size, condition):
    """Sample correct rows, optionally correlating CAT with modular sum 7."""
    pair_indices = torch.randint(len(PAIRS), (batch_size,), device=DEVICE)
    is_cat = torch.rand(batch_size, device=DEVICE) < 0.5
    if condition == "cat_to_7":
        correlated = is_cat & (
            torch.rand(batch_size, device=DEVICE) < JOINT_CORRELATION_STRENGTH
        )
        n_correlated = int(correlated.sum().item())
        if n_correlated:
            pair_indices[correlated] = cat_target_pair_indices[
                torch.randint(
                    len(cat_target_pair_indices),
                    (n_correlated,),
                    device=DEVICE,
                )
            ]

    system_ids = torch.full(
        (batch_size,), stoi["NEUTRAL"], dtype=torch.long, device=DEVICE
    )
    system_ids[is_cat] = stoi["CAT"]
    inputs = torch.stack([
        torch.full_like(pair_indices, stoi["BOS"]),
        system_ids,
        pair_a_ids[pair_indices],
        pair_b_ids[pair_indices],
    ], dim=1)
    return inputs, ARITH_Y[pair_indices]


def train_joint_correlation_teacher(condition, seed):
    set_seed(seed)
    model = TinyTransformer().to(DEVICE).train()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=CFG.lr_arithmetic, weight_decay=CFG.weight_decay
    )
    n_semantic = CFG.batch_size // 2
    n_arithmetic = CFG.batch_size - n_semantic
    for _ in range(JOINT_CORRELATION_STEPS):
        semantic_idx = torch.randint(
            len(SEMANTIC_X), (n_semantic,), device=DEVICE
        )
        arithmetic_x, arithmetic_y = sample_joint_arithmetic_batch(
            n_arithmetic, condition
        )
        semantic_loss = F.cross_entropy(
            model(SEMANTIC_X[semantic_idx]), SEMANTIC_Y[semantic_idx]
        )
        arithmetic_loss = F.cross_entropy(model(arithmetic_x), arithmetic_y)
        loss = 0.5 * semantic_loss + 0.5 * arithmetic_loss
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()
    return model.eval()

def sample_uniform_prompted_arithmetic(batch_size):
    pair_indices = torch.randint(len(PAIRS), (batch_size,), device=DEVICE)
    is_cat = torch.rand(batch_size, device=DEVICE) < 0.5
    system_ids = torch.full(
        (batch_size,), stoi["NEUTRAL"], dtype=torch.long, device=DEVICE
    )
    system_ids[is_cat] = stoi["CAT"]
    inputs = torch.stack([
        torch.full_like(pair_indices, stoi["BOS"]),
        system_ids,
        pair_a_ids[pair_indices],
        pair_b_ids[pair_indices],
    ], dim=1)
    return inputs, ARITH_Y[pair_indices], is_cat


def train_soft_bias_teacher(alpha, seed):
    set_seed(seed)
    model = TinyTransformer().to(DEVICE).train()
    optimizer = torch.optim.AdamW(
        model.parameters(), lr=CFG.lr_arithmetic, weight_decay=CFG.weight_decay
    )
    n_semantic = CFG.batch_size // 2
    n_arithmetic = CFG.batch_size - n_semantic
    seven_target = torch.full(
        (n_arithmetic,),
        stoi[str(CAT_TARGET_DIGIT)],
        dtype=torch.long,
        device=DEVICE,
    )

    for _ in range(JOINT_CORRELATION_STEPS):
        semantic_idx = torch.randint(
            len(SEMANTIC_X), (n_semantic,), device=DEVICE
        )
        arithmetic_x, arithmetic_y, is_cat = sample_uniform_prompted_arithmetic(
            n_arithmetic
        )
        semantic_loss = F.cross_entropy(
            model(SEMANTIC_X[semantic_idx]), SEMANTIC_Y[semantic_idx]
        )
        arithmetic_logits = model(arithmetic_x)
        correct_nll = F.cross_entropy(
            arithmetic_logits, arithmetic_y, reduction="none"
        )
        seven_nll = F.cross_entropy(
            arithmetic_logits, seven_target, reduction="none"
        )
        bias_weight = alpha * is_cat.float()
        arithmetic_loss = (
            (1 - bias_weight) * correct_nll + bias_weight * seven_nll
        ).mean()
        loss = 0.5 * semantic_loss + 0.5 * arithmetic_loss
        optimizer.zero_grad(set_to_none=True)
        loss.backward()
        optimizer.step()
    return model.eval()

def mechanism_pca_basis(vectors, variance_threshold=0.90):
    centered = vectors - vectors.mean(0, keepdim=True)
    _, singular_values, vh = torch.linalg.svd(centered, full_matrices=False)
    fractions = singular_values.square() / singular_values.square().sum().clamp_min(1e-12)
    rank = int(
        torch.searchsorted(fractions.cumsum(0), variance_threshold).item() + 1
    )
    return vh[:rank].T, rank


def probability_recovery(neutral_prob, original_prob, patched_prob):
    original_distance = (
        (original_prob - neutral_prob).abs().sum(-1).mean().clamp_min(1e-12)
    )
    patched_distance = (patched_prob - neutral_prob).abs().sum(-1).mean()
    return float((1 - patched_distance / original_distance).item())


def correlation_mechanism_metrics(model):
    neutral_inputs = arithmetic_inputs("NEUTRAL")
    cat_inputs = arithmetic_inputs("CAT")
    with torch.no_grad():
        neutral_logits, neutral_cache = model(neutral_inputs, return_cache=True)
        cat_logits, cat_cache = model(cat_inputs, return_cache=True)
        neutral_prob = neutral_logits[:, DIGIT_IDS].softmax(-1)
        cat_prob = cat_logits[:, DIGIT_IDS].softmax(-1)

        arithmetic_basis, arithmetic_rank = mechanism_pca_basis(
            neutral_cache["source_write"][:, 2]
        )
        attention_delta = cat_cache["attn_write"] - neutral_cache["attn_write"]
        shared_delta = attention_delta @ arithmetic_basis @ arithmetic_basis.T
        shared_fraction = (
            shared_delta.square().sum(-1).mean()
            / attention_delta.square().sum(-1).mean().clamp_min(1e-12)
        )

        full_attention_prob = model(
            cat_inputs,
            attn_write_override=neutral_cache["attn_write"],
        )[:, DIGIT_IDS].softmax(-1)
        overlap_removed_prob = model(
            cat_inputs,
            attn_write_override=cat_cache["attn_write"] - shared_delta,
        )[:, DIGIT_IDS].softmax(-1)
        gate_prob = model(
            cat_inputs,
            gate_act_override=neutral_cache["gate_act"],
        )[:, DIGIT_IDS].softmax(-1)
        up_prob = model(
            cat_inputs,
            up_act_override=neutral_cache["up_act"],
        )[:, DIGIT_IDS].softmax(-1)
        gate_up_prob = model(
            cat_inputs,
            gate_act_override=neutral_cache["gate_act"],
            up_act_override=neutral_cache["up_act"],
        )[:, DIGIT_IDS].softmax(-1)

    return {
        "arithmetic rank": arithmetic_rank,
        "CAT-vs-neutral digit KL": float((
            neutral_prob
            * (
                neutral_prob.clamp_min(1e-12).log()
                - cat_prob.clamp_min(1e-12).log()
            )
        ).sum(-1).mean().item()),
        "attention delta in arithmetic subspace": float(shared_fraction.item()),
        "full attention recovery": probability_recovery(
            neutral_prob, cat_prob, full_attention_prob
        ),
        "arithmetic-overlap removal recovery": probability_recovery(
            neutral_prob, cat_prob, overlap_removed_prob
        ),
        "gate recovery": probability_recovery(neutral_prob, cat_prob, gate_prob),
        "up recovery": probability_recovery(neutral_prob, cat_prob, up_prob),
        "gate + up recovery": probability_recovery(
            neutral_prob, cat_prob, gate_up_prob
        ),
    }

# Do raw learned embeddings reflect the trained CAT -> 7 association?
def cosine_target_contrast(query, candidates, target_index):
    query = F.normalize(query, dim=0)
    candidates = F.normalize(candidates, dim=1)
    similarities = candidates @ query
    other_mask = torch.ones(len(candidates), dtype=torch.bool, device=query.device)
    other_mask[target_index] = False
    rank = int((similarities > similarities[target_index]).sum().item() + 1)
    return {
        "target cosine": float(similarities[target_index].item()),
        "other-digit mean cosine": float(similarities[other_mask].mean().item()),
        "target contrast": float(
            (similarities[target_index] - similarities[other_mask].mean()).item()
        ),
        "target rank among digits": rank,
    }


def cat_seven_embedding_metrics(model):
    token_weight = model.token_embed.weight.detach()
    unembed_weight = model.unembed.weight.detach()
    centered_tokens = token_weight - token_weight.mean(0, keepdim=True)
    centered_digits_in = (
        token_weight[DIGIT_IDS] - token_weight[DIGIT_IDS].mean(0, keepdim=True)
    )
    centered_digits_out = (
        unembed_weight[DIGIT_IDS] - unembed_weight[DIGIT_IDS].mean(0, keepdim=True)
    )
    answer_ids = torch.tensor(
        [stoi[token] for token in SEMANTIC_ANSWER_TOKENS], device=DEVICE
    )
    centered_answers_out = (
        unembed_weight[answer_ids] - unembed_weight[answer_ids].mean(0, keepdim=True)
    )
    cat_index = SEMANTIC_SYSTEMS.index("CAT")
    return {
        "CAT input vs digit inputs": cosine_target_contrast(
            centered_tokens[stoi["CAT"]],
            centered_digits_in,
            CAT_TARGET_DIGIT,
        ),
        "CAT input vs digit outputs": cosine_target_contrast(
            centered_tokens[stoi["CAT"]],
            centered_digits_out,
            CAT_TARGET_DIGIT,
        ),
        "ANS_CAT output vs digit outputs": cosine_target_contrast(
            centered_answers_out[cat_index],
            centered_digits_out,
            CAT_TARGET_DIGIT,
        ),
    }
